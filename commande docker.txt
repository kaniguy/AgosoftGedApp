docker images                 : Lister les images
docker ps                     : Conteneurs actifs
docker ps -a                  : Tous les conteneurs
docker pull image             : Télécharger une image
docker build -t nom .         : Construire une image
docker run image              : Créer un conteneur
docker start container        : Démarrer
docker stop container         : Arrêter
docker restart container      : Redémarrer
docker rm container           : Supprimer un conteneur
docker rmi image              : Supprimer une image
docker logs container         : Voir les logs
docker exec -it container cmd : Entrer dans un conteneur
docker compose up -d          : Démarrer un projet Compose
docker compose down           : Arrêter un projet Compose





docker builder prune -af        : Cache de build (images intermédiaires)
docker image prune -af          : Images non utilisées (dangling)
docker container prune -f       : Conteneurs arrêtés
docker volume prune -f          : Volumes non utilisés
docker network prune -f         : Réseaux non utilisés
docker system prune -af         : TOUT en une fois

docker images                 : Lister les images disponibles
docker pull image:tag         : Télécharger une image
docker build -t nom-image .   : Construire une image
docker rmi nom-image          : Supprimer une image
docker tag image nom:tag      : Donner un nouveau tag à une image
docker save -o image.tar image:tag : Exporter une image
docker load -i image.tar      : Importer une image


docker ps                     : Voir les conteneurs en cours
docker ps -a                  : Voir tous les conteneurs
docker run image              : Créer et démarrer un conteneur
docker run -d image           : Démarrer en arrière-plan
docker start nom-container    : Démarrer un conteneur arrêté
docker stop nom-container     : Arrêter un conteneur
docker restart nom-container  : Redémarrer un conteneur
docker rm nom-container       : Supprimer un conteneur
docker logs nom-container     : Voir les logs
docker exec -it nom-container cmd : Ouvrir un CMD dans le conteneur

docker port nom-container     : Voir les ports utilisés
docker run -p 8080:80 image   : Mapper le port 8080 vers le port 80

docker --version              : Afficher la version de Docker
docker info                   : Afficher les informations sur Docker
docker help                   : Afficher l'aide Docker


docker volume ls              : Lister les volumes
docker volume inspect nom     : Voir les détails d'un volume
docker volume create nom      : Créer un volume
docker volume rm nom          : Supprimer un volume
docker volume prune -f        : Supprimer les volumes inutilisés


docker network ls             : Lister les réseaux
docker network inspect nom    : Voir les détails d'un réseau
docker network create nom     : Créer un réseau
docker network rm nom        : Supprimer un réseau
docker network prune -f       : Supprimer les réseaux inutilisés

docker compose up -d           : Démarrer tous les services
docker compose down            : Arrêter et supprimer les services
docker compose ps              : Voir l'état des services
docker compose logs            : Voir les logs
docker compose logs -f         : Suivre les logs en temps réel
docker compose restart         : Redémarrer les services
docker compose stop            : Arrêter les services
docker compose start           : Démarrer les services arrêtés
docker compose pull            : Télécharger les dernières images
docker compose build          : Construire les images
docker compose up -d --build   : Reconstruire et démarrer


docker login                   : Se connecter à Docker Hub
docker logout                  : Se déconnecter
docker push utilisateur/image  : Envoyer une image sur Docker Hub
docker pull utilisateur/image  : Télécharger une image
